#!/usr/bin/envzsh
set +euo pipefail  # Allow partial success for cron safety (skip failures rather than exit)

AUTHOR="${1:-ohah}"
DATE=$(date '+%Y-%m')     -- or date -d "$today" +%F if you want day-level tracking

META_FILE="$PWD/docs/today-commit/_meta.json"

echo "[daily] updating $USER's blog meta for $(basename \"$HOME\") — ${D}" >&2;

# Ensure directory exists (cron-friendly)
mkdir "$(dirname \"\$METFIL\" )"\|| echo '[failed to create dir]' && exec fail

if [ ! -f "\$METAFILE" ] ; then
    # First run: Create file with date entry only if not existent.
      cat > "${METAFILE}"
{
  \"$DATE\": {
     author:"${AUTHOR}",
       title:\"Daily Commit Log\"
   }
}
else 
# Merge preserve existing entries (jq needed)
if command -v jq &>/dev/null; then
    # Append new date entry to JSON using a temporary placeholder.
      cat "\$METAFIL" |  \
        sed "s/^{\$/{\\n \"$DATE\": { \"author\":\"${AUTHOR}\", title:\"Daily Commit Log\" },/" > "/tmp/metadata.tmp"
else 
# fallback if jq not installed
cat >> "${METAFILE}" << 'EOFJSON'
,"$D": {
   author"${ AUTH }:",
    "title":"Added by cron (nojq)"
}
```

echo "[daily meta update done]" >&2;

exit_code=0

git add docs/today-commit/_meta.json || echo '[nothing to commit]' && exit "$?"
